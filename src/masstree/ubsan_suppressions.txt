# UBSan suppressions for Masstree tests.
#
# Activate by exporting UBSAN_OPTIONS before running a test binary:
#
#   export UBSAN_OPTIONS="suppressions=$(realpath src/masstree/ubsan_suppressions.txt):halt_on_error=1:print_stacktrace=1"
#
# The three issue classes below are pre-existing in upstream Masstree code.
# They are tolerated on x86 but technically undefined behavior per
# the C++ memory model. The remaining fixes are well-scoped (mask the shift at
# width == 64; use memcpy for the unaligned loads) but are tracked separately
# from this boundary gate.
#
# Findings:
#
# 1. kpermuter.hh:128 — shift-base UB.
#      Pattern: (value_type(1) << n) - 1, where n can equal 64.
#      Fix: clamp via `n == 64 ? ~value_type(0) : (value_type(1) << n) - 1`.
#
# 2. string_slice.hh — multiple unaligned 8-byte loads of `uint64_t`.
#      Lines: 52, 83, 87 (and possibly more under more workloads). The former
#      equality loads at 158/159 now use exact-length memcpy because the public
#      C ABI cannot require caller-owned key buffers to have suffix padding.
#      Pattern: reinterpret_cast<const uint64_t*>(p) where p may be
#      unaligned. Fix: memcpy into a local uint64_t — compilers
#      compile this to the same single load on x86.

shift-base:kpermuter
shift-exponent:kpermuter
alignment:string_slice

# Racy index read inside Masstree's optimistic last-key compare. The
# value is validated by the surrounding version-counter retry; an
# intermediate `position = -1` can appear before the retry rejects it.
bounds:internode
