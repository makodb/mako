# UBSan suppressions for Masstree tests.
#
# Activate by exporting UBSAN_OPTIONS before running a test binary:
#
#   export UBSAN_OPTIONS="suppressions=$(realpath src/masstree/ubsan_suppressions.txt):halt_on_error=1:print_stacktrace=1"
#
# The former kpermuter shift-width and string_slice unaligned-load findings are
# fixed in-tree. Do not suppress them: UBSan must catch any regression.

# Known out-of-bounds UB inside Masstree's optimistic last-key compare. A racy
# size read can produce `position = -1`; the surrounding version retry discards
# the value but does not make the intermediate C++ access language-defined.
bounds:internode
